```sql

         SELECT fact_work_order.work_order_id,
            fact_work_order.number,
            fact_work_order.team_id,
            fact_work_order.technician_key,
            fact_work_order.customer_key,

          WHERE fact_work_order.created_at_utc IS NOT NULL

 SELECT count(*) AS total_work_orders,
    count(*) FILTER (WHERE completed_at_utc IS NOT NULL) AS completed_work_orders,
    count(*) FILTER (WHERE completed_at_utc IS NOT NULL)::double precision / NULLIF(count(*), 0)::double precision AS completion_rate,
    ( SELECT count(*) FILTER (WHERE s.completed_at_utc IS NOT NULL AND NOT s.breached)::double precision / NULLIF(count(*) FILTER (WHERE s.completed_at_utc IS NOT NULL), 0)::double precision
           FROM warehouse.fact_workflow_stage s

             JOIN eligible o USING (work_order_id)
          WHERE s.completed_at_utc IS NOT NULL) AS average_stage_minutes,
    NULL::double precision AS technician_utilization,

           FROM warehouse.fact_sla_breach b
             JOIN eligible o USING (work_order_id))) / NULLIF(count(*), 0)::double precision AS escalation_frequency,
    (( SELECT count(DISTINCT b.work_order_id)::double precision AS count
           FROM warehouse.fact_booking b
             JOIN eligible o USING (work_order_id)
          WHERE b.status = 'Confirmed'::text)) / NULLIF(count(*), 0)::double precision AS scheduling_efficiency,
    max(last_ingested_at_utc) AS last_ingested_at_utc
   FROM eligible;
```

```sql
WITH eligible AS (
         SELECT fact_work_order.work_order_id,
            fact_work_order.number,
            fact_work_order.team_id,
            fact_work_order.technician_key,
            fact_work_order.customer_key,
            fact_work_order.category_key,
            fact_work_order.status_key,
            fact_work_order.created_date_key,
            fact_work_order.priority,
            fact_work_order.created_at_utc,
            fact_work_order.completed_at_utc,
            fact_work_order.last_occurrence_utc,
            fact_work_order.last_ingested_at_utc,
            fact_work_order.order_source_sequence,
            fact_work_order.order_source_occurred_at_utc,
            fact_work_order.source_status,
            fact_work_order.workflow_started_at_utc
           FROM warehouse.fact_work_order
          WHERE fact_work_order.created_at_utc IS NOT NULL
        ), orders AS (
         SELECT eligible.team_id,
            count(*) AS total_work_orders,
            count(*) FILTER (WHERE eligible.completed_at_utc IS NOT NULL) AS completed_work_orders,
            avg(EXTRACT(epoch FROM eligible.completed_at_utc - eligible.created_at_utc) / 60.0) FILTER (WHERE eligible.completed_at_utc IS NOT NULL) AS average_service_minutes,
            max(eligible.last_ingested_at_utc) AS last_ingested_at_utc
           FROM eligible
          GROUP BY eligible.team_id
        ), stages AS (
         SELECT o_1.team_id,
            count(*) FILTER (WHERE s_1.completed_at_utc IS NOT NULL) AS completed_stages,
            count(*) FILTER (WHERE s_1.completed_at_utc IS NOT NULL AND NOT s_1.breached) AS compliant_stages,
            avg(s_1.effective_minutes) FILTER (WHERE s_1.completed_at_utc IS NOT NULL) AS average_stage_minutes,
            count(*) FILTER (WHERE s_1.breached) AS breaches
           FROM analytics.v_stage_duration s_1
             JOIN eligible o_1 USING (work_order_id)
          GROUP BY o_1.team_id
        ), bookings AS (
         SELECT o_1.team_id,
            count(DISTINCT b_1.work_order_id) FILTER (WHERE b_1.status = 'Confirmed'::text) AS assigned_orders
           FROM warehouse.fact_booking b_1
             JOIN eligible o_1 USING (work_order_id)
          GROUP BY o_1.team_id
        )
 SELECT o.team_id,
    o.total_work_orders,
    o.completed_work_orders,
    o.average_service_minutes,
    o.last_ingested_at_utc,
    o.completed_work_orders::double precision / NULLIF(o.total_work_orders, 0)::double precision AS completion_rate,
    s.compliant_stages::double precision / NULLIF(s.completed_stages, 0)::double precision AS sla_compliance_rate,
    s.average_stage_minutes,
    s.breaches::double precision / NULLIF(o.total_work_orders, 0)::double precision AS escalation_frequency,
    b.assigned_orders::double precision / NULLIF(o.total_work_orders, 0)::double precision AS scheduling_efficiency,
    NULL::double precision AS technician_utilization
   FROM orders o
     LEFT JOIN stages s USING (team_id)
     LEFT JOIN bookings b USING (team_id);
```
